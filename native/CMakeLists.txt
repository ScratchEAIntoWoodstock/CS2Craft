cmake_minimum_required(VERSION 3.25)
project(CS2CraftNative LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
add_subdirectory(../third_party/minhook ${CMAKE_BINARY_DIR}/minhook)
add_library(CS2Craft.Renderer SHARED Overlay.cpp WorldRenderer.cpp WorldRendererTest.cpp InputRouter.cpp)
target_compile_definitions(CS2Craft.Renderer PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(CS2Craft.Renderer PRIVATE minhook d3d11 dxgi d3dcompiler user32)
target_compile_options(CS2Craft.Renderer PRIVATE /W4 /EHsc)
add_executable(CS2Craft.RendererTest RendererTest.cpp)
target_compile_definitions(CS2Craft.RendererTest PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(CS2Craft.RendererTest PRIVATE CS2Craft.Renderer)
